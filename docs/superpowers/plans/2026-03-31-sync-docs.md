```python
#!/usr/bin/env python3
"""sync-docs: 플러그인 README/CLAUDE.md 자동 동기화."""
import json
import re
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
PLUGINS = ["harness", "flutter-toolkit", "design-kit"]


def parse_frontmatter(path: Path) -> dict | None:
    """SKILL.md / agent .md 파일에서 YAML frontmatter를 파싱한다.

    표준 라이브러리만 사용 (PyYAML 없이 간이 파싱).
    """
    text = path.read_text(encoding="utf-8")
    m = re.match(r"^---\s*\n(.*?)\n---", text, re.DOTALL)
    if not m:
        return None
    fm: dict[str, str] = {}
    current_key = None
    current_val_lines: list[str] = []
    for line in m.group(1).splitlines():
        # key: value
        kv = re.match(r"^(\w[\w-]*):\s*(.*)", line)
        if kv:
            if current_key:
                fm[current_key] = "\n".join(current_val_lines).strip()
            current_key = kv.group(1)
            val = kv.group(2)
            # handle > (block scalar)
            if val.strip() == ">":
                current_val_lines = []
            else:
                current_val_lines = [val.strip()]
        elif current_key and line.startswith("  "):
            current_val_lines.append(line.strip())
    if current_key:
        fm[current_key] = "\n".join(current_val_lines).strip()
    return fm


def first_line(text: str) -> str:
    """여러 줄 description에서 첫 줄만 추출."""
    lines = text.strip().splitlines()
    return lines[0].strip() if lines else ""


def replace_marker(content: str, marker: str, replacement: str) -> str:
    """<!-- AUTO:marker --> ... <!-- /AUTO:marker --> 사이를 교체."""
    pattern = re.compile(
        rf"(<!-- AUTO:{re.escape(marker)} -->)\n.*?\n(<!-- /AUTO:{re.escape(marker)} -->)",
        re.DOTALL,
    )
    return pattern.sub(rf"\1\n{replacement}\n\2", content)


def has_marker(content: str, marker: str) -> bool:
    return f"<!-- AUTO:{marker} -->" in content
```

```python
def collect_skills(plugin_dir: Path) -> list[dict]:
    """skills/*/SKILL.md에서 스킬 목록 수집."""
    skills_dir = plugin_dir / "skills"
    if not skills_dir.exists():
        return []
    result = []
    for skill_dir in sorted(skills_dir.iterdir()):
        skill_md = skill_dir / "SKILL.md"
        if not skill_md.exists():
            continue
        fm = parse_frontmatter(skill_md)
        if fm is None:
            print(f"[sync-docs] 경고: frontmatter 없음 — {skill_md}", file=sys.stderr)
            continue
        result.append({
            "name": fm.get("name", skill_dir.name),
            "description": first_line(fm.get("description", "")),
        })
    return result


def collect_agents(plugin_dir: Path) -> list[dict]:
    """agents/*.md에서 에이전트 목록 수집."""
    agents_dir = plugin_dir / "agents"
    if not agents_dir.exists():
        return []
    result = []
    for agent_md in sorted(agents_dir.glob("*.md")):
        fm = parse_frontmatter(agent_md)
        if fm is None:
            print(f"[sync-docs] 경고: frontmatter 없음 — {agent_md}", file=sys.stderr)
            continue
        result.append({
            "name": fm.get("name", agent_md.stem),
            "description": first_line(fm.get("description", "")),
            "model": fm.get("model", ""),
            "tools": fm.get("tools", ""),
        })
    return result


def collect_hooks(plugin_dir: Path) -> list[dict]:
    """hooks/hooks.json에서 훅 목록 수집."""
    hooks_file = plugin_dir / "hooks" / "hooks.json"
    if not hooks_file.exists():
        return []
    data = json.loads(hooks_file.read_text(encoding="utf-8"))
    result = []
    for event, entries in data.get("hooks", {}).items():
        for entry in entries:
            for hook in entry.get("hooks", []):
                cmd = hook.get("command", "")
                # ${CLAUDE_PLUGIN_ROOT}/scripts/foo.sh → scripts/foo.sh
                cmd_short = re.sub(r".*scripts/", "scripts/", cmd)
                result.append({
                    "event": event,
                    "command": cmd_short,
                    "matcher": entry.get("matcher", ""),
                })
    return result


def collect_scripts(plugin_dir: Path) -> list[dict]:
    """scripts/*.sh에서 스크립트 목록 + 설명 주석 수집."""
    scripts_dir = plugin_dir / "scripts"
    if not scripts_dir.exists():
        return []
    result = []
    for sh in sorted(scripts_dir.glob("*.sh")):
        desc = ""
        for line in sh.read_text(encoding="utf-8").splitlines()[1:5]:
            m = re.match(r"^#\s*[─—]*\s*(.+?)\s*[─—]*\s*$", line)
            if m:
                desc = m.group(1)
                break
        result.append({"name": sh.name, "description": desc})
    return result


def collect_evals(plugin_dir: Path) -> list[dict]:
    """evals/ 디렉토리 내 파일 목록 수집 (1단계만)."""
    evals_dir = plugin_dir / "evals"
    if not evals_dir.exists():
        return []
    result = []
    for f in sorted(evals_dir.iterdir()):
        if f.name.startswith("."):
            continue
        result.append({"name": f.name, "is_dir": f.is_dir()})
    return result


def collect_references(plugin_dir: Path) -> list[dict]:
    """references/*.md에서 레퍼런스 목록 수집."""
    refs_dir = plugin_dir / "references"
    if not refs_dir.exists():
        return []
    result = []
    for md in sorted(refs_dir.glob("*")):
        if md.name.startswith("."):
            continue
        desc = ""
        if md.is_file():
            fm = parse_frontmatter(md)
            if fm and "description" in fm:
                desc = first_line(fm["description"])
            else:
                # 첫 번째 # 헤더 찾기
                for line in md.read_text(encoding="utf-8").splitlines()[:10]:
                    hm = re.match(r"^#\s+(.+)", line)
                    if hm:
                        desc = hm.group(1)
                        break
        result.append({"name": md.name, "description": desc})
    return result


def load_plugin_json(plugin_dir: Path) -> dict:
    pj = plugin_dir / ".claude-plugin" / "plugin.json"
    if not pj.exists():
        return {}
    return json.loads(pj.read_text(encoding="utf-8"))


def load_marketplace() -> dict:
    mp = ROOT / ".claude-plugin" / "marketplace.json"
    return json.loads(mp.read_text(encoding="utf-8"))
```

```python
def render_skills_table(skills: list[dict]) -> str:
    lines = ["| 스킬 | 설명 |", "|------|------|"]
    for s in skills:
        lines.append(f"| `{s['name']}` | {s['description']} |")
    return "\n".join(lines)


def render_agents_table(agents: list[dict]) -> str:
    lines = ["| 에이전트 | 모델 | 도구 | 설명 |", "|----------|------|------|------|"]
    for a in agents:
        lines.append(f"| `{a['name']}` | {a['model']} | {a['tools']} | {a['description']} |")
    return "\n".join(lines)


def render_hooks_table(hooks: list[dict]) -> str:
    lines = ["| 이벤트 | 매처 | 실행 |", "|--------|------|------|"]
    for h in hooks:
        matcher = h["matcher"] or "—"
        lines.append(f"| {h['event']} | {matcher} | `{h['command']}` |")
    return "\n".join(lines)


def render_scripts_table(scripts: list[dict]) -> str:
    lines = ["| 스크립트 | 설명 |", "|----------|------|"]
    for s in scripts:
        lines.append(f"| `{s['name']}` | {s['description']} |")
    return "\n".join(lines)


def render_evals_table(evals: list[dict]) -> str:
    lines = ["| 파일 | 유형 |", "|------|------|"]
    for e in evals:
        kind = "디렉토리" if e["is_dir"] else "파일"
        lines.append(f"| `{e['name']}` | {kind} |")
    return "\n".join(lines)


def render_references_table(refs: list[dict]) -> str:
    lines = ["| 파일 | 설명 |", "|------|------|"]
    for r in refs:
        lines.append(f"| `{r['name']}` | {r['description']} |")
    return "\n".join(lines)


def sync_plugin_readme(plugin_name: str, dry_run: bool = False) -> list[str]:
    """플러그인 README의 AUTO 마커 섹션을 갱신한다. 변경된 마커 목록 반환."""
    plugin_dir = ROOT / plugin_name
    readme_path = plugin_dir / "README.md"
    if not readme_path.exists():
        print(f"[sync-docs] 경고: {readme_path} 없음, 스킵", file=sys.stderr)
        return []

    content = readme_path.read_text(encoding="utf-8")
    original = content
    pj = load_plugin_json(plugin_dir)
    version = pj.get("version", "?")
    changed_markers: list[str] = []

    # 헤더 갱신: # Name · vX.Y.Z 또는 # Name (첫 줄)
    header_pattern = re.compile(r"^#\s+.+$", re.MULTILINE)
    header_match = header_pattern.match(content)
    if header_match:
        display_name = plugin_name.replace("-", " ").title()
        if plugin_name == "flutter-toolkit":
            display_name = "Flutter Toolkit"
        elif plugin_name == "design-kit":
            display_name = "design-kit"
        elif plugin_name == "harness":
            display_name = "Harness Engineering"
        new_header = f"# {display_name} · v{version}"
        content = content[:header_match.start()] + new_header + content[header_match.end():]

    marker_map = {
        "skills": render_skills_table(collect_skills(plugin_dir)),
        "agents": render_agents_table(collect_agents(plugin_dir)),
        "hooks": render_hooks_table(collect_hooks(plugin_dir)),
        "scripts": render_scripts_table(collect_scripts(plugin_dir)),
        "evals": render_evals_table(collect_evals(plugin_dir)),
        "references": render_references_table(collect_references(plugin_dir)),
    }

    for marker, table in marker_map.items():
        if has_marker(content, marker):
            new_content = replace_marker(content, marker, table)
            if new_content != content:
                changed_markers.append(marker)
            content = new_content

    if content != original:
        if dry_run:
            print(f"[sync-docs] (dry-run) {plugin_name}/README.md — {', '.join(changed_markers) or '헤더'} 갱신")
        else:
            readme_path.write_text(content, encoding="utf-8")
            print(f"[sync-docs] {plugin_name}/README.md — {', '.join(changed_markers) or '헤더'} 갱신")
    return changed_markers
```

```python
def sync_root_readme(dry_run: bool = False) -> bool:
    """루트 README.md의 AUTO:plugins 마커를 갱신한다."""
    readme_path = ROOT / "README.md"
    content = readme_path.read_text(encoding="utf-8")
    original = content

    if not has_marker(content, "plugins"):
        print("[sync-docs] 경고: README.md에 AUTO:plugins 마커 없음, 스킵", file=sys.stderr)
        return False

    mp = load_marketplace()
    lines = ["| 플러그인 | 버전 | 스킬 수 | 설명 |", "|----------|------|---------|------|"]
    for p in mp.get("plugins", []):
        pname = p["name"]
        plugin_dir = ROOT / pname
        pj = load_plugin_json(plugin_dir)
        version = pj.get("version", "?")
        skill_count = len(collect_skills(plugin_dir))
        desc = pj.get("description", "")
        lines.append(f"| [`{pname}`](./{pname}/) | v{version} | {skill_count}종 | {desc} |")

    content = replace_marker(content, "plugins", "\n".join(lines))

    if content != original:
        if dry_run:
            print("[sync-docs] (dry-run) README.md — 플러그인 테이블 갱신")
        else:
            readme_path.write_text(content, encoding="utf-8")
            print("[sync-docs] README.md — 플러그인 테이블 갱신")
        return True
    return False


def sync_claude_md(dry_run: bool = False) -> bool:
    """CLAUDE.md의 AUTO:summary 마커를 갱신한다."""
    claude_md = ROOT / "CLAUDE.md"
    content = claude_md.read_text(encoding="utf-8")
    original = content

    if not has_marker(content, "summary"):
        print("[sync-docs] 경고: CLAUDE.md에 AUTO:summary 마커 없음, 스킵", file=sys.stderr)
        return False

    lines = []
    for pname in PLUGINS:
        plugin_dir = ROOT / pname
        pj = load_plugin_json(plugin_dir)
        version = pj.get("version", "?")
        skill_count = len(collect_skills(plugin_dir))
        agent_count = len(collect_agents(plugin_dir))
        desc = pj.get("description", "")
        parts = [f"v{version}"]
        if skill_count:
            parts.append(f"스킬 {skill_count}종")
        if agent_count:
            parts.append(f"에이전트 {agent_count}종")
        lines.append(f"- **{pname}** ({', '.join(parts)}) — {desc}")

    content = replace_marker(content, "summary", "\n".join(lines))

    if content != original:
        if dry_run:
            print("[sync-docs] (dry-run) CLAUDE.md — 요약 갱신")
        else:
            claude_md.write_text(content, encoding="utf-8")
            print("[sync-docs] CLAUDE.md — 요약 갱신")
        return True
    return False
```

파일 끝에 추가:

```python
def main() -> None:
    args = sys.argv[1:]
    dry_run = "--dry-run" in args
    check_only = "--check-only" in args
    args = [a for a in args if not a.startswith("--")]

    target_plugins = args if args else PLUGINS

    # 유효하지 않은 플러그인 이름 검사
    for p in target_plugins:
        if p not in PLUGINS:
            print(f"[sync-docs] 에러: 알 수 없는 플러그인 '{p}'. 가능: {', '.join(PLUGINS)}", file=sys.stderr)
            sys.exit(1)

    if check_only:
        # 현재 README 내용과 생성될 내용을 비교하여 동기화 필요 여부만 출력
        needs_sync = False
        for pname in target_plugins:
            plugin_dir = ROOT / pname
            readme_path = plugin_dir / "README.md"
            if not readme_path.exists():
                continue
            content = readme_path.read_text(encoding="utf-8")
            # 스킬 수 비교로 간단히 판단
            skills = collect_skills(plugin_dir)
            agents = collect_agents(plugin_dir)
            for marker, items in [("skills", skills), ("agents", agents)]:
                if has_marker(content, marker):
                    # 마커 안 내용 추출
                    m = re.search(
                        rf"<!-- AUTO:{marker} -->\n(.*?)\n<!-- /AUTO:{marker} -->",
                        content, re.DOTALL
                    )
                    if m:
                        current_rows = len([l for l in m.group(1).splitlines() if l.startswith("|") and not l.startswith("|---") and not l.startswith("| 스") and not l.startswith("| 에")])
                        if current_rows != len(items):
                            print(f"[sync-docs] 동기화 필요: {pname} {marker} ({current_rows} → {len(items)})")
                            needs_sync = True

        if not needs_sync:
            print("[sync-docs] 모든 문서가 최신 상태입니다.")
        else:
            print(f"[sync-docs] `python scripts/sync-docs.py{' ' + ' '.join(target_plugins) if args else ''}` 를 실행하세요.")
        return

    changed_count = 0

    for pname in target_plugins:
        markers = sync_plugin_readme(pname, dry_run=dry_run)
        if markers:
            changed_count += 1

    # 루트 README + CLAUDE.md는 항상 갱신 (특정 플러그인 지정해도)
    if sync_root_readme(dry_run=dry_run):
        changed_count += 1
    if sync_claude_md(dry_run=dry_run):
        changed_count += 1

    if changed_count:
        print(f"[sync-docs] {changed_count}개 파일 갱신 완료")
    else:
        print("[sync-docs] 변경 사항 없음")


if __name__ == "__main__":
    main()
```
