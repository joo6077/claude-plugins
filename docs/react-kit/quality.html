tests/unit/data/repositories/user-repository.test.ts

import { describe, it, expect, beforeAll, afterEach, afterAll } from 'vitest'
import { setupServer } from 'msw/node'
import { http, HttpResponse } from 'msw'
import { userRepository } from '@/data/repositories/user-repository'

const server = setupServer()
beforeAll(() => server.listen())
afterEach(() => server.resetHandlers())
afterAll(() => server.close())

describe('userRepository.fetchUser', () => {
  it('returns ok(User) on 200', async () => {
    server.use(
      http.get('*/users/123', () =>
        HttpResponse.json({
          id: '123',
          email_address: 'jane@example.com',
          display_name: 'Jane',
          created_at: '2026-04-10T00:00:00Z',
        }),
      ),
    )
    const result = await userRepository.fetchUser('123')
    expect(result.isOk()).toBe(true)
    if (result.isOk()) {
      expect(result.value.email).toBe('jane@example.com')
    }
  })

  it('returns err(not-found) on 404', async () => {
    server.use(http.get('*/users/999', () => new HttpResponse(null, { status: 404 })))
    const result = await userRepository.fetchUser('999')
    expect(result.isErr()).toBe(true)
    if (result.isErr()) {
      expect(result.error.kind).toBe('user/not-found')
    }
  })
})

import { describe, it, expect, vi } from 'vitest'
import { render, screen } from '@testing-library/react'
import userEvent from '@testing-library/user-event'
import { ok, err } from 'neverthrow'
import { LoginForm } from '@/presentation/features/auth/components/login-form'

describe('LoginForm', () => {
  it('shows validation error when email is empty', async () => {
    const user = userEvent.setup()
    const onSubmit = vi.fn().mockResolvedValue(ok(undefined))
    render(<LoginForm onSubmit={onSubmit} />)

    await user.click(screen.getByRole('button', { name: /로그인/i }))
    expect(await screen.findByText(/올바른 이메일/i)).toBeInTheDocument()
    expect(onSubmit).not.toHaveBeenCalled()
  })

  it('calls onSubmit with values and shows server error on Result.err', async () => {
    const user = userEvent.setup()
    const onSubmit = vi.fn().mockResolvedValue(err({ kind: 'login/invalid-credentials' as const }))
    render(<LoginForm onSubmit={onSubmit} />)

    await user.type(screen.getByLabelText(/이메일/i), 'jane@example.com')
    await user.type(screen.getByLabelText(/비밀번호/i), 'password123')
    await user.click(screen.getByRole('button', { name: /로그인/i }))

    expect(onSubmit).toHaveBeenCalledWith({ email: 'jane@example.com', password: 'password123' })
    expect(await screen.findByText(/이메일 또는 비밀번호가 올바르지 않습니다/i)).toBeInTheDocument()
  })
})

1.7 에러 경로 테스트 패턴

Result 패턴에는 throw 가 없다. 그래서 전통적인 expect(() => ...).toThrow() 가 아니라 isErr() 확인으로 검증한다.

Result 에 맞지 않는 검증
// Result 를 돌려주는 함수는 throw 하지 않는다 — 이 단언은 에러 경로를 못 본다
expect(() => userRepository.fetchUser('123')).toThrow()
isErr() 로 검증
it('returns invalid-format Failure when API returns bad shape', async () => {
  server.use(
    http.get('*/users/123', () => HttpResponse.json({ wrong: 'shape' })),
  )
  const result = await userRepository.fetchUser('123')
  expect(result.isErr()).toBe(true)
  if (result.isErr()) {
    expect(result.error.kind).toBe('user/validation-failed')
    expect(result.error.issues.length).toBeGreaterThan(0)
  }
})

Error Boundary 테스트 — 일부러 throw 하는 자식으로 경계를 건드린다

1.8 /react-test 주의사항

  • !vitest.config 에 jsdom 환경 명시 — component 테스트 파일은 jsdom, unit 테스트는 node. environmentMatchGlobs 로 경로마다 나누기를 권장한다.
  • !setupFiles 필수 — @testing-library/jest-dom matchers 를 전역 등록하려면 setup 파일에서 expect.extend(matchers) 를 실행한다.
  • !cleanup() 자동 — Testing Library 는 테스트마다 DOM 을 자동으로 정리하지만, 글로벌 모드에서는 afterEach(cleanup) 를 적어 두기를 권장한다.
  • !Playwright 와 Vitest 함께 쓰기 — 둘 다 "test" 라는 이름을 쓴다. 설정은 playwright.config.ts 와 vitest.config.ts 로 나누고, 파일 확장자도 .spec.ts (Playwright) 와 .test.ts (Vitest) 로 가르기를 권장한다.
  • !fetch 모의는 MSW 우선 — vi.stubGlobal('fetch', ...) 같은 저수준 목은 피하고 MSW 로 한다. 요청 모양이 문서로 남는 효과도 있다.
  • !MSW v2 API 변경 — 구 rest.get(...) 가 아니라 http.get(...) + HttpResponse. deprecated API 는 쓰지 않는다.
  • !비동기 단언을 잊기 쉽다 — findBy* 는 반드시 await 한다. 잊으면 pending Promise 때문에 문제가 있는데도 테스트가 통과한다.
  • !Strict TS — vi.mocked(...) 로 모킹한 객체의 타입 안전성을 지킨다. as any 금지.

2.4 Step 1 — 데이터 경계에서 throw → Failure 변환

G2 /react-api 가 이미 이 패턴을 만든다. 다시 보면:

src/data/datasources/remote/user-api.ts

export function fetchUserDto(id: string): ResultAsync<unknown, UserFailure> {
  return ResultAsync.fromPromise(
    fetch(`${BASE}/users/${id}`).then(async (r) => {
      if (r.status === 404) throw { kind: 'user/not-found', userId: id }
      if (!r.ok) throw { kind: 'user/network-error', cause: r.statusText }
      return r.json()
    }),
    (e) => (isUserFailure(e)
      ? e
      : { kind: 'user/network-error', cause: String(e) }),
  )
}
출처: neverthrow GitHub (ResultAsync.fromPromise)

규칙

1
throw 는 datasource 안에서만 한다. 곧바로 ResultAsync.fromPromise 의 두 번째 인자에서 Failure 로 바꾼다.
2
Failure 는 discriminated union 이다. kind 필드로 분기한다.
3
cause 필드에 원본 에러 메시지를 남긴다 (디버깅용).

2.5 Step 2 — Failure 전파 + Severity 매핑

src/domain/failures/user-failures.ts

export type Severity = 'info' | 'warning' | 'error' | 'fatal'

export type UserFailure =
  | { kind: 'user/not-found'; userId: string }
  | { kind: 'user/unauthorized' }
  | { kind: 'user/network-error'; cause: string }
  | { kind: 'user/validation-failed'; issues: string[] }
  | { kind: 'user/rate-limited'; retryAfterSec: number }

export function severityOf(failure: UserFailure): Severity {
  switch (failure.kind) {
    case 'user/not-found':
      return 'info'       // 사용자 인지 충분
    case 'user/rate-limited':
      return 'warning'    // 일시적, 재시도 안내
    case 'user/unauthorized':
    case 'user/validation-failed':
      return 'error'      // 사용자 액션 필요
    case 'user/network-error':
      return 'fatal'      // 네트워크 자체 문제
  }
}

Severity 는 사용자 관점의 심각도다

기술적 에러 레벨이 아니다. "네트워크 없음" 은 기술적으로는 warning 급이지만 사용자에게는 fatal 이다.

kindSeverity이유
user/not-foundinfo사용자 인지 충분
user/rate-limitedwarning일시적, 재시도 안내
user/unauthorizederror사용자 액션 필요
user/validation-failederror사용자 액션 필요
user/network-errorfatal네트워크 자체 문제

type FailureLike = { kind: string; cause?: string }

export function displayFailure(failure: FailureLike, severity: Severity): void {

    case 'info':

      break
    case 'warning':

      break

      break
  }
}

2.7 React Error Boundary

Result 패턴이 못 덮는 것은 React 렌더 중 throw 다. 이것을 Error Boundary 가 잡는다.

src/presentation/shared/components/root-error-boundary.tsx

import * as React from 'react'

type Props = { children: React.ReactNode }
type State = { hasError: boolean; error: Error | null }

export class RootErrorBoundary extends React.Component<Props, State> {
  state: State = { hasError: false, error: null }

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error }
  }

  componentDidCatch(error: Error, errorInfo: React.ErrorInfo): void {
    console.error('RootErrorBoundary:', error, errorInfo)
    // 프로덕션: Sentry 같은 서비스로 전송
  }

  render(): React.ReactNode {
    if (this.state.hasError) {
      return (
        <div role="alert" className="flex min-h-screen items-center justify-center">
          <div className="space-y-4 text-center">
            <h1 className="text-2xl font-semibold">예기치 못한 오류</h1>
            <p className="text-muted-foreground">{this.state.error?.message}</p>
            <button onClick={() => window.location.reload()}>새로고침</button>
          </div>
        </div>
      )
    }
    return this.props.children
  }
}

Error Boundary 는 최후의 안전망이다. 1차 에러 처리는 Result 로 한다. Error Boundary 가 자주 열리면 Result 패턴에 구멍이 있다는 신호다.

2.8 /react-error 주의사항

금지 패턴
// 금지: 비동기 에러 — Error Boundary 에 닿지 않는다
useEffect(() => {
  fetch('/api/users').then(() => {
    throw new Error('parse failed')
  })
}, [])

// 금지: Failure 를 Error 로 감싸기 — stack trace 가 무의미해지고 타입이 사라진다
throw new Error(JSON.stringify(failure))

// 금지: domain 에서 throw — throw 는 datasource 안에서만
throw new Error('not found')
권장 패턴
// 권장: 비동기 경로를 ResultAsync 로 감싼다
const result = await ResultAsync.fromPromise(
  fetch('/api/users').then((r) => r.json()),
  (e) => ({ kind: 'user/network-error', cause: String(e) }),
)

// 권장: Failure 값을 그대로 전파한다
return err({ kind: 'user/not-found', userId: id })

// 권장: never 로 모든 kind 를 다뤘는지 검증한다
function assertNever(x: never): never {
  throw new Error(`Unhandled: ${JSON.stringify(x)}`)
}
  • !비동기 에러는 Error Boundary 가 못 잡는다 — useEffect 안의 throw, Promise reject 는 Error Boundary 에 닿지 않는다. 반드시 Result 로 감싼다.
  • !Error Boundary 는 클래스 컴포넌트 필수 — React 19 에서도 아직 hooks 로 만들 수 없다. react-error-boundary 같은 라이브러리는 옵션이고, 기본은 자체 구현이다.
  • !Toast 라이브러리 선택 — shadcn/ui 는 sonner 를 권장한다. Radix 기반 <Toaster /> 와 겹치지 않게 하나만 고른다.
  • !fatal 상태 — Zustand store 에 fatalError: Failure | null 필드를 두고 최상위 컴포넌트에서 렌더를 가른다.
  • !개발 환경에서 Boundary 를 건너뛰는 것처럼 보인다 — Vite dev 모드에서는 Error Boundary 가 렌더 에러를 못 잡는 것처럼 보일 수 있다 (React 가 dev 에서 에러를 다시 던져 개발자에게 보여 준다). 프로덕션 빌드로 확인한다.
  • !Failure 를 Error 로 감싸지 않는다 — Failure 는 그 자체로 타입 안전한 값이다. new Error(JSON.stringify(failure)) 같은 변환은 stack trace 를 무의미하게 만들고 타입 정보를 잃는다.
  • !Strict TS — Failure discriminated union 의 모든 kind 를 switch 에서 다뤘는지 never 타입으로 검증한다.

3.5 번역 추가 + codegen 흐름

/react-l10n 이 "로그인" 이라는 한국어 문자열을 추가할 때의 순서다.

1
대상 컴포넌트에 매크로를 넣는다 (<Trans>로그인</Trans> 또는 t`로그인`).
2
소스에 import 를 더한다 (import { Trans } from '@lingui/react/macro').
3
추출 명령 pnpm lingui extract — 코드에서 매크로를 훑어 en.po 에 새 키를 더한다.
4
각 locale 파일 (ko.po, ja.po) 에 빈 번역 항목이 자동으로 생긴다.
5
사용자에게 "ko.po 에 번역 추가하세요" 라고 안내한다 — 번역 자체는 사람이 입력한다.
6
compile: pnpm lingui compile — .po → .js 컴파일된 catalog 를 만든다 (Vite 플러그인이 자동으로 한다).

CLI 명령

pnpm lingui extract                   # 소스 → .po 동기화
pnpm lingui compile                   # .po → runtime catalog
pnpm lingui extract --clean           # 안 쓰는 키 정리 — 기본 흐름 아님, 사용자가 요청할 때만 (react-l10n §4-1)

--clean 은 기본 흐름에 넣지 않는다 — 사용자가 요청할 때만 돌린다. --clean 은 소스에서 더는 찾지 못한 메시지를 catalog 에서 지운다. 매크로가 적용되지 않아 extractor 가 메시지를 못 찾으면 번역이 채워진 항목까지 함께 지워진다. 옵션 자체는 정상이라 금지하지는 않는다.

  1. 번역 폴더에 커밋하지 않은 변경이 없는지 먼저 본다. 있으면 알리고 멈춘다.
  2. pnpm lingui extract --clean 을 돌리고 바로 locales 폴더의 git diff --stat 을 보인다.
  3. 지워진 키 수와, 번역이 채워져 있다가 지워진 항목을 따로 센다.
  4. 채워진 번역이 지워졌으면 그 목록을 보이고, 사용자가 확인하기 전에는 커밋하지 않는다.

3.7 /react-l10n 주의사항

1매크로 import 경로 혼동

v5 에서 @lingui/macro (구버전) 가 @lingui/react/macro (JSX/React) 와 @lingui/core/macro (core) 로 나뉘었다. 잘못 import 하면 컴파일 단계에서 매크로가 적용되지 않는다.

2msgid 대신 해시 id 가 기본

v5 는 코드의 원본 문자열을 해시해 id 를 자동으로 만든다. id 를 직접 정하려면 <Trans id="custom.id" /> 로 적는다.

3번역 누락

extract 뒤 ko.po 에 빈 msgstr 로 남은 키는 런타임에 원본 (en) 으로 돌아간다. /react-audit 이 "빈 번역 키" 수로 경고한다.

4plurals, interpolation

<Plural>, <Select> 같은 전용 매크로를 쓴다. {count} 개 처럼 단순 치환은 <Trans> 안에서 {count} 그대로 둔다.

5Vite 플러그인 순서

react({ plugins: [['@lingui/swc-plugin', {}]] }) 가 반드시 lingui 플러그인보다 앞에 온다. SWC 가 매크로를 먼저 transpile 한 뒤 lingui 플러그인이 catalog 를 넣는다.

6Strict TS

매크로는 타입을 잘 추론한다. 동적 key 는 msg`...` 로 감싸고 i18n._(msg) 로 부른다. any 금지.

7서버 컴포넌트는 쓰지 않는다

react-kit 기본 구성 (Vite + TanStack Router) 은 SSR 이 없다. Lingui 의 RSC 지원은 여기와 관련이 없다.

3.8 Clean Architecture 배치

  • 번역 매크로 사용src/presentation/**/*.tsx — presentation 레이어만
  • locale catalogsrc/infrastructure/i18n/locales/<locale>.po + 컴파일된 .ts
  • i18n setupsrc/infrastructure/i18n/setup.ts
  • configlingui.config.ts (프로젝트 루트)

절대 금지 — domain/ 에 매크로를 import 하지 않는다. 도메인 레이어는 i18n 을 모른다. Failure 의 사용자 메시지는 presentation 레이어 (display-failure.ts) 에서 매핑한다.
