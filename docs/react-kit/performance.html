// src/data/datasources/wasm/image-worker.ts
import { expose } from 'comlink'
import init, { resize_thumbnail } from '@/wasm/core'

let ready: Promise<void> | null = null
function ensureInit(): Promise<void> {
  if (!ready) ready = init().then(() => undefined)
  return ready
}

export const imageWorkerApi = {
  async resizeThumbnail(input: Uint8Array, width: number, height: number): Promise<Uint8Array> {
    await ensureInit()
    return resize_thumbnail(input, width, height)
  },
}

UseCase — neverthrow ResultAsync 래핑

// src/domain/usecases/image-usecases.ts
import { ResultAsync } from 'neverthrow'
import { resizeImageViaWasm } from '@/data/datasources/wasm/image-client'

export type ImageFailure =
  | { kind: 'image/wasm-not-ready'; cause: string }
  | { kind: 'image/resize-failed'; cause: string }

export function resizeImage(
  input: Uint8Array,
  size: { width: number; height: number },
): ResultAsync<Uint8Array, ImageFailure> {
  return ResultAsync.fromPromise(
    resizeImageViaWasm(input, size),
    (e) => ({
      kind: 'image/resize-failed' as const,
      cause: e instanceof Error ? e.message : String(e),
    }),
  )
}

/react-tauri — Tauri command/event 브릿지

Tauri 2 Rust 백엔드 command를 정의하고 React에서 invoke하는 브릿지를 자동 생성. 반드시 isTauri() feature detection 가드를 포함한다.

[1] Rust command 정의 — src-tauri/src/commands/config.rs

#[tauri::command]
pub fn read_config_file(path: String) -> Result<AppConfig, String> {
    let content = fs::read_to_string(&path)
        .map_err(|e| format!("failed to read {}: {}", path, e))?;
    serde_json::from_str::<AppConfig>(&content)
        .map_err(|e| format!("invalid config format: {}", e))
}

            commands::config::read_config_file,
        ])
        .run(tauri::generate_context!())
        .expect("error while running tauri application");
}

[4] TS 래퍼 — src/infrastructure/tauri/config.ts

import { invoke, isTauri } from '@tauri-apps/api/core'
import { z } from 'zod'
import { ok, err, type Result } from 'neverthrow'

const AppConfigSchema = z.object({ theme: z.string(), language: z.string() })
export type AppConfig = z.infer<typeof AppConfigSchema>

export type ReadConfigFailure =
  | { kind: 'config/not-in-tauri' }
  | { kind: 'config/read-failed'; cause: string }
  | { kind: 'config/invalid-format'; issues: string[] }

export async function readConfigFile(path: string): Promise<Result<AppConfig, ReadConfigFailure>> {
  if (!isTauri()) return err({ kind: 'config/not-in-tauri' })
  try {
    const raw = await invoke<unknown>('read_config_file', { path })
    const parsed = AppConfigSchema.safeParse(raw)
    if (!parsed.success) return err({ kind: 'config/invalid-format', issues: parsed.error.issues.map(i => i.message) })
    return ok(parsed.data)
  } catch (e) {
    return err({ kind: 'config/read-failed', cause: e instanceof Error ? e.message : String(e) })
  }
}
isTauri() 가드 없이 Tauri API를 직접 import하면 웹 빌드에서 런타임 오류. 모든 Tauri 호출은 infrastructure/tauri/ 래퍼를 경유해야 하며, presentation/이나 domain/에서 직접 import 금지.

Rust panic → Result 변환 경로

1️⃣

Rust 쪽 — panic 대신 Result

가능하면 Result<T, JsError> 반환으로 panic 대신 에러 값 반환 우선. ? 연산자 적극 활용.

2️⃣

console_error_panic_hook

어쩔 수 없는 panic (무한 재귀, 배열 OOB): console_error_panic_hook 크레이트로 JS 콘솔에 메시지 출력.

3️⃣

Worker 경계 포획

Comlink는 throw를 Promise reject로 전달. ResultAsync.fromPromise(..., (e) => Failure)로 모든 throw를 Failure로 포획.

4️⃣

최종 보장

UseCase 레벨에서는 panic 포함 항상 Result<T, Failure>를 반환. domain 레이어는 throw를 절대 보지 않는다.

WASM 클라이언트

Worker 클라이언트 싱글톤 패턴

Comlink wrap으로 Worker를 감싸고, 싱글톤으로 관리한다. 매 호출마다 Worker를 생성하면 Spawn 비용 누적 + 메모리 누수가 발생한다. 

// src/data/datasources/wasm/image-client.ts
import { wrap, type Remote } from 'comlink'
import type { imageWorkerApi } from './image-worker'

let workerInstance: Remote<typeof imageWorkerApi> | null = null

function getClient(): Remote<typeof imageWorkerApi> {
  if (workerInstance) return workerInstance
  const worker = new Worker(new URL('./image-worker.ts', import.meta.url), {
    type: 'module',
  })
  workerInstance = wrap<typeof imageWorkerApi>(worker)
  return workerInstance
}

export async function resizeImageViaWasm(
  input: Uint8Array,
  size: { width: number; height: number },
): Promise<Uint8Array> {
  const client = getClient()
  return client.resizeThumbnail(input, size.width, size.height)
}

isTauri() 가드 — 웹/데스크탑 분기

React 코드는 웹 배포 + Tauri 데스크탑 배포 양쪽에서 동일하게 동작해야 한다. 런타임 feature detection으로 분기한다. 

// isTauri() 가드 패턴 — 항상 이 방식으로
import { isTauri } from '@tauri-apps/api/core'

export async function loadConfig(path: string) {
  if (!isTauri()) {
    // 웹 환경 — localStorage 또는 fetch fallback
    return err({ kind: 'config/not-in-tauri' })
  }
  // Tauri 환경 — 네이티브 파일 접근
  return readConfigFile(path)
}

안티패턴 비교

Bad vs Good — G3 핵심 실수 패턴

g3-performance.md §1.5 / §1.6 / §2.6 Gotchas 기반. WASM·Tauri 통합에서 반복 관찰된 오류를 코드 수준으로 비교한다.

❌ 안티패턴 — render 안에서 WASM 직접 호출
function ImageProcessor({ file }: Props) {
  // 매 렌더마다 WASM 경계 비용 발생
  const resized = wasmResize(file.buffer, 800, 600)
  return <img src={URL.createObjectURL(resized)} />
}

렌더마다 JS→WASM 경계를 넘는다. /react-audit §8 감사 항목에서 error로 감지된다.

✅ 권장 — useMemo / useMutation으로 분리
function ImageProcessor({ file }: Props) {
  const { mutate, data } = useMutation({
    mutationFn: (buf: ArrayBuffer) =>
      wasmClient.resize(buf, 800, 600), // Comlink Worker
  })
  useEffect(() => { mutate(file.buffer) }, [file])
  return data ? <img src={URL.createObjectURL(data)} /> : <Spinner />
}

WASM 호출은 Comlink Worker + useMutation으로 메인 스레드 블로킹 없이 실행한다.

❌ 안티패턴 — Tauri API 직접 import
// presentation/ 또는 domain/ 에서 직접 사용
import { readTextFile } from '@tauri-apps/plugin-fs'

async function loadConfig(path: string) {
  return readTextFile(path) // 웹 빌드에서 런타임 오류
}

웹 빌드(브라우저)에서 실행 시 런타임 오류. /react-audit AP-02 규칙이 error 레벨로 감지한다.

✅ 권장 — infrastructure/tauri/ 래퍼 + isTauri() 가드
// src/infrastructure/tauri/fs.ts
import { isTauri } from '@tauri-apps/api/core'
import { readTextFile } from '@tauri-apps/plugin-fs'
import { ok, err } from 'neverthrow'

export async function readConfig(path: string) {
  if (!isTauri()) return err({ kind: 'NotSupported' })
  const text = await readTextFile(path)
  return ok(text)
}

isTauri() 런타임 가드로 웹/데스크탑 양쪽 dist를 단일 번들로 공유할 수 있다.

❌ 안티패턴 — Rust에서 panic! 전파
#[wasm_bindgen]
pub fn parse_csv(data: &[u8]) -> Vec<Row> {
    let text = std::str::from_utf8(data).unwrap(); // panic!
    csv::Reader::from_reader(text.as_bytes())
        .records()
        .map(|r| r.unwrap()) // panic!
        .collect()
}

WASM panic은 JS 쪽에서 unhandled exception으로 나타나 에러 경로 추적이 불가능하다.

✅ 권장 — Result<T, JsError> 반환
use wasm_bindgen::JsError;

#[wasm_bindgen]
pub fn parse_csv(data: &[u8]) -> Result<JsValue, JsError> {
    let text = std::str::from_utf8(data)
        .map_err(|e| JsError::new(&e.to_string()))?;
    let rows: Vec<Row> = csv::Reader::from_reader(text.as_bytes())
        .records()
        .collect::<Result<_, _>>()
        .map_err(|e| JsError::new(&e.to_string()))?;
    Ok(serde_wasm_bindgen::to_value(&rows)?)
}

Result 반환으로 JS 쪽이 정상 try/catch 또는 neverthrow Result로 처리 가능해진다.
