OpenAPI 원칙

utoipa 5.x를 사용하여 Rust 코드에서 OpenAPI 3.x 스펙을 생성한다. 핸들러 함수와 DTO에 매크로를 붙이면 런타임에 JSON 스펙이 자동 생성되고, Swagger UI로 서빙한다.

핵심 원칙

utoipa를 사용한 코드 우선 OpenAPI 스펙 생성 원칙 4가지.

1

ToSchema + #[utoipa::path]

DTO에 ToSchema를 derive하고, 핸들러에 #[utoipa::path]를 붙인다. 요청 바디, 응답 타입, 상태 코드, 보안 스키마를 선언적으로 정의한다.

#[derive(Serialize, Deserialize, ToSchema)]
pub struct CreateUserRequest {
    /// 사용자 이메일 주소
    pub email: String,
    /// 사용자 이름 (2~50자)
    pub name: String,
}

#[utoipa::path(
    post,
    path = "/users",
    request_body = CreateUserRequest,
    responses(
        (status = 201, description = "생성 성공", body = UserResponse),
        (status = 400, description = "잘못된 요청", body = ErrorResponse),
        (status = 409, description = "이미 존재하는 이메일"),
    ),
    tag = "users",
    security(("bearer_auth" = [])),
)]
pub async fn create_user(...) -> impl IntoResponse { ... }
utoipa 5.x
2

#[derive(OpenApi)]로 스펙 루트 정의

앱 최상위에서 #[derive(OpenApi)]로 모든 path와 schema를 등록한다. Bearer 인증 스키마는 Modify trait으로 런타임에 주입한다.

#[derive(OpenApi)]
#[openapi(
    paths(users::create_user, users::get_user, users::list_users),
    components(schemas(
        CreateUserRequest, UserResponse, ErrorResponse,
    )),
    tags((name = "users", description = "사용자 관리 API")),
    modifiers(&SecurityAddon),
    info(title = "My API", version = "1.0.0"),
)]
pub struct ApiDoc;

struct SecurityAddon;
impl Modify for SecurityAddon {
    fn modify(&self, openapi: &mut utoipa::openapi::OpenApi) {
        let components = openapi.components
            .get_or_insert_with(Default::default);
        components.add_security_scheme(
            "bearer_auth",
            SecurityScheme::Http(
                HttpBuilder::new()
                    .scheme(HttpAuthScheme::Bearer)
                    .bearer_format("JWT")
                    .build(),
            ),
        );
    }
}
컴파일 타임 스펙 생성
3

Swagger UI 마운트

Swagger UI를 axum 라우터에 .merge()로 마운트한다. 프로덕션에서는 환경 변수로 조건부 비활성화한다.

// Cargo.toml
utoipa = { version = "5", features = ["axum_extras"] }
utoipa-swagger-ui = { version = "9", features = ["axum"] }

// router 설정
pub fn create_router(state: AppState) -> Router {
    let mut router = Router::new()
        .nest("/users", users::router())
        .with_state(state);

    if std::env::var("ENABLE_SWAGGER_UI").is_ok() {
        router = router.merge(
            SwaggerUi::new("/swagger-ui")
                .url("/api-docs/openapi.json", ApiDoc::openapi())
        );
    }
    router
}
/swagger-ui (조건부)
4

모듈별 OpenApi merge

대규모 앱에서 paths()에 모든 핸들러를 나열하면 유지보수가 어렵다. 각 모듈에 서브 OpenApi를 정의하고 merge로 합친다.

// users 모듈
#[derive(OpenApi)]
#[openapi(
    paths(create_user, get_user),
    components(schemas(UserResponse)),
)]
struct UsersApiDoc;

// posts 모듈
#[derive(OpenApi)]
#[openapi(
    paths(create_post, get_post),
    components(schemas(PostResponse)),
)]
struct PostsApiDoc;

// 최상위에서 머지
let mut doc = ApiDoc::openapi();
doc.merge(UsersApiDoc::openapi());
doc.merge(PostsApiDoc::openapi());
ApiDoc + UsersApiDoc + PostsApiDoc = merged spec

안티패턴

utoipa 사용 시 반복되는 실수와 올바른 대안.

Bad — 한 파일에 모든 핸들러 나열
#[derive(OpenApi)]
#[openapi(
    paths(
        // 50개 핸들러를 모두 여기에...
        users::create_user,
        users::get_user,
        posts::create_post,
        posts::get_post,
        // ... 유지보수 불가 ❌
    ),
)]
pub struct ApiDoc;
Good — 모듈별 분리 + merge
// 각 모듈이 자체 OpenApi 보유
// users/mod.rs
#[derive(OpenApi)]
#[openapi(paths(create_user, get_user))]
struct UsersApiDoc;

// 최상위에서 merge
let mut doc = ApiDoc::openapi();
doc.merge(UsersApiDoc::openapi()); // ✅
Bad — ToSchema와 serde rename 불일치
#[derive(Serialize, Deserialize, ToSchema)]
#[serde(rename_all = "camelCase")]
pub struct UserResponse {
    pub user_id: String,  // JSON: "userId"
    // ToSchema는 "user_id"로 스펙 생성 ❌
    // 스펙과 실제 JSON 불일치
}
Good — schema rename_all 명시
#[derive(Serialize, Deserialize, ToSchema)]
#[serde(rename_all = "camelCase")]
#[schema(rename_all = "camelCase")] // ✅ 스펙도 동기화
pub struct UserResponse {
    pub user_id: String,  // 스펙: "userId" ✅
}
Bad — 프로덕션에서 Swagger UI 무조건 노출
// 환경 분리 없이 항상 마운트
Router::new()
    .merge(SwaggerUi::new("/swagger-ui")
        .url("/api-docs/openapi.json", ApiDoc::openapi()))
    // 내부 API 스펙 외부 노출 ❌
Good — 환경변수 조건부 마운트
if std::env::var("ENABLE_SWAGGER_UI").is_ok() {
    router = router.merge(
        SwaggerUi::new("/swagger-ui")
            .url("/api-docs/openapi.json", ApiDoc::openapi())
    );
} // 프로덕션에서 비활성화 ✅
Gotchas 체크리스트

utoipa 작업 시 반복적으로 실수하는 포인트.

  • ✓ axum_extras feature 필수 — axum의 Path, Query extractor를 utoipa가 인식하려면 features = ["axum_extras"]를 활성화해야 한다. 없으면 extractor 파라미터가 스펙에 반영되지 않는다.
  • ✓ Modify로만 런타임 스펙 변경 가능 — #[derive(OpenApi)]는 컴파일 타임에 스펙을 생성한다. 서버 URL이나 보안 스키마처럼 환경에 따라 달라지는 값은 Modify 구현체를 통해 런타임에 주입한다.
  • ✓ openapi.json 엔드포인트는 GET만 지원 — SwaggerUi::url()이 등록하는 엔드포인트는 GET만 처리한다. 별도 인증이 필요하면 .merge() 이전에 미들웨어를 적용하거나 스펙 엔드포인트를 직접 구현한다.
  • ✓ 제네릭 타입 ToSchema 제한 — PageResponse<T>처럼 제네릭 타입은 #[schema(value_type = ...)]으로 구체 타입을 지정하거나, 사용되는 모든 구체 타입을 components에 등록해야 한다.
  • ✓ doc comment가 description이 된다 — DTO 필드 위의 /// ... 주석이 OpenAPI description 필드로 변환된다. 필드 설명을 주석으로 작성하면 별도 #[schema(description = ...)] 없이도 스펙에 반영된다.
